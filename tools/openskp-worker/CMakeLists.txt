cmake_minimum_required(VERSION 3.21)
project(GreenLensOpenSkpDevelopmentWorker VERSION 0.1.0 LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "This experimental worker is restricted to native Windows/MSVC")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(GREENLENS_PREPARED_SOURCE "" CACHE PATH "New exact source tree made by prepare-source.mjs")
if(NOT EXISTS "${GREENLENS_PREPARED_SOURCE}/source-manifest.json" OR EXISTS "${GREENLENS_PREPARED_SOURCE}/PREPARATION_FAILED.txt")
  message(FATAL_ERROR "Missing or failed prepared source manifest; no automatic fetch permitted")
endif()
file(READ "${GREENLENS_PREPARED_SOURCE}/source-manifest.json" source_manifest)
string(JSON manifest_schema GET "${source_manifest}" schema_version)
if(NOT manifest_schema STREQUAL "greenlens.openskp-development-source.v1")
  message(FATAL_ERROR "Unsupported prepared-source contract")
endif()
string(JSON nfiles LENGTH "${source_manifest}" files)
math(EXPR last_file "${nfiles}-1")
foreach(i RANGE 0 ${last_file})
  string(JSON relative GET "${source_manifest}" files ${i} path)
  string(JSON expected GET "${source_manifest}" files ${i} prepared_sha256)
  if(relative MATCHES "(^/|\\\\|\.\.)")
    message(FATAL_ERROR "Invalid prepared source path")
  endif()
  file(SHA256 "${GREENLENS_PREPARED_SOURCE}/${relative}" actual)
  if(NOT actual STREQUAL expected)
    message(FATAL_ERROR "Prepared source changed: ${relative}")
  endif()
endforeach()
file(SHA256 "${GREENLENS_PREPARED_SOURCE}/openskp/src/strict_limits.hpp" limits_actual)
string(JSON limits_expected GET "${source_manifest}" strict_limits_sha256)
if(NOT limits_actual STREQUAL limits_expected)
  message(FATAL_ERROR "Prepared strict limits changed")
endif()
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/include/openskp")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/include/openskp/export.hpp" "#pragma once\n#define OPENSKP_EXPORT\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/include/miniz_export.h" "#pragma once\n#define MINIZ_EXPORT\n")
set(P "${GREENLENS_PREPARED_SOURCE}/openskp")
set(M "${GREENLENS_PREPARED_SOURCE}/miniz")
set(T "${GREENLENS_PREPARED_SOURCE}/tinygltf")
set(units core earcut errors face_groups geometry instanced_glb instanced_scene legacy model observability pages_dimensions tlv transforms triangulator vff)
set(sources)
foreach(unit IN LISTS units)
  list(APPEND sources "${P}/src/${unit}.cpp")
endforeach()
add_library(greenlens_openskp_strict STATIC ${sources} "${M}/miniz.c" "${M}/miniz_tdef.c" "${M}/miniz_tinfl.c" "${M}/miniz_zip.c")
target_include_directories(greenlens_openskp_strict PUBLIC "${P}/include" "${CMAKE_CURRENT_BINARY_DIR}/include" PRIVATE "${P}/src" "${M}" "${T}")
target_compile_definitions(greenlens_openskp_strict PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
target_compile_options(greenlens_openskp_strict PRIVATE /utf-8 /W3)
add_executable(greenlens-openskp-development-worker worker.cpp)
target_include_directories(greenlens-openskp-development-worker PRIVATE "${P}/src" "${T}")
target_compile_definitions(greenlens-openskp-development-worker PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(greenlens-openskp-development-worker PRIVATE /utf-8 /W4)
target_link_libraries(greenlens-openskp-development-worker PRIVATE greenlens_openskp_strict bcrypt)
# No install target, shared SDK, network acquisition, extra exporter or app dependency.
