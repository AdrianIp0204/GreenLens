cmake_minimum_required(VERSION 3.21)
project(GreenLensPreallocationUnitTests LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "These source-bound unit tests require Windows/MSVC")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(GREENLENS_PREALLOCATION_SOURCE "" CACHE PATH "Prepared miniz/helper unit-test source")
if(NOT EXISTS "${GREENLENS_PREALLOCATION_SOURCE}/unit-source-manifest.json" OR
   EXISTS "${GREENLENS_PREALLOCATION_SOURCE}/PREPARATION_FAILED.txt")
  message(FATAL_ERROR "Use prepare-preallocation-tests.mjs; no automatic acquisition")
endif()
file(READ "${GREENLENS_PREALLOCATION_SOURCE}/unit-source-manifest.json" manifest)
string(JSON schema GET "${manifest}" schema_version)
if(NOT schema STREQUAL "greenlens.openskp-preallocation-unit-source.v1")
  message(FATAL_ERROR "Unsupported unit-source manifest")
endif()
string(JSON count LENGTH "${manifest}" files)
math(EXPR last "${count}-1")
foreach(i RANGE 0 ${last})
  string(JSON relative GET "${manifest}" files ${i} path)
  string(JSON expected GET "${manifest}" files ${i} sha256)
  if(relative MATCHES "(^/|\\\\|\\.\\.)")
    message(FATAL_ERROR "Invalid unit-source path")
  endif()
  file(SHA256 "${GREENLENS_PREALLOCATION_SOURCE}/${relative}" actual)
  if(NOT actual STREQUAL expected)
    message(FATAL_ERROR "Unit source changed: ${relative}")
  endif()
endforeach()
set(P "${GREENLENS_PREALLOCATION_SOURCE}")
add_executable(greenlens-preallocation-tests "${P}/preallocation_test.cpp"
  "${P}/miniz.c" "${P}/miniz_tdef.c" "${P}/miniz_tinfl.c" "${P}/miniz_zip.c")
target_include_directories(greenlens-preallocation-tests PRIVATE "${P}")
target_compile_definitions(greenlens-preallocation-tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
target_compile_options(greenlens-preallocation-tests PRIVATE /utf-8 /W3)
# This target contains no OpenSKP parser/worker, model input or image exporter.
